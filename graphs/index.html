1. Enter Vertices and Edges and Represent Them in an Adjacency Matrix


#include <stdio.h>

int main() {
    int a[20][20], n, e, i, j, u, v;

    printf("Enter number of vertices: ");
    scanf("%d", &n);

    if (n <= 0 || n > 20) return 0;

    for (i = 0; i < n; i++)
        for (j = 0; j < n; j++)
            a[i][j] = 0;

    printf("Enter number of edges: ");
    scanf("%d", &e);

    printf("Enter edges (vertices numbered 0 to %d):\n", n - 1);

    for (i = 0; i < e; i++) {
        scanf("%d%d", &u, &v);

        if (u >= 0 && u < n && v >= 0 && v < n) {
            a[u][v] = 1;
            a[v][u] = 1;
        }
    }

    printf("Adjacency Matrix:\n");

    for (i = 0; i < n; i++) {
        for (j = 0; j < n; j++)
            printf("%d ", a[i][j]);
        printf("\n");
    }

    return 0;
}

2. Create a Graph as an Adjacency List with Insertion and Deletion


#include <stdio.h>
#include <stdlib.h>

struct node {
    int vertex;
    struct node *next;
};

struct node *adj[20];
int n;

int insertEdge(int u, int v) {
    struct node *p;

    if (u < 0 || u >= n || v < 0 || v >= n)
        return 0;

    p = malloc(sizeof(struct node));
    if (p == NULL) exit(1);

    p->vertex = v;
    p->next = adj[u];
    adj[u] = p;

    p = malloc(sizeof(struct node));
    if (p == NULL) exit(1);

    p->vertex = u;
    p->next = adj[v];
    adj[v] = p;

    return 1;
}

int deleteOne(int u, int v) {
    struct node *p = adj[u], *prev = NULL;

    while (p != NULL) {
        if (p->vertex == v) {
            if (prev == NULL)
                adj[u] = p->next;
            else
                prev->next = p->next;

            free(p);
            return 1;
        }

        prev = p;
        p = p->next;
    }

    return 0;
}

int deleteEdge(int u, int v) {
    int x, y;

    if (u < 0 || u >= n || v < 0 || v >= n)
        return 0;

    x = deleteOne(u, v);
    y = deleteOne(v, u);

    return x && y;
}

int display() {
    int i;
    struct node *p;

    for (i = 0; i < n; i++) {
        printf("%d: ", i);

        p = adj[i];
        while (p != NULL) {
            printf("%d -> ", p->vertex);
            p = p->next;
        }

        printf("NULL\n");
    }

    return 1;
}

int main() {
    int i, choice, u, v;

    printf("Enter number of vertices (max 20): ");
    scanf("%d", &n);

    if (n <= 0 || n > 20) return 0;

    for (i = 0; i < n; i++)
        adj[i] = NULL;

    do {
        printf("\n1.Insert Edge\n2.Delete Edge\n3.Display\n4.Exit\n");
        printf("Enter choice: ");
        scanf("%d", &choice);

        if (choice == 1 || choice == 2) {
            printf("Enter vertices u and v: ");
            scanf("%d%d", &u, &v);

            if (choice == 1) {
                if (insertEdge(u, v))
                    printf("Edge inserted\n");
                else
                    printf("Invalid vertices\n");
            } else {
                if (deleteEdge(u, v))
                    printf("Edge deleted\n");
                else
                    printf("Edge not found\n");
            }
        } else if (choice == 3) {
            display();
        }

    } while (choice != 4);

    return 0;
}

3. Traverse an Adjacency Matrix Using Depth First Search (DFS)


#include <stdio.h>

int a[20][20], visited[20], n;

int dfs(int v) {
    int i;

    visited[v] = 1;
    printf("%d ", v);

    for (i = 0; i < n; i++) {
        if (a[v][i] == 1 && visited[i] == 0)
            dfs(i);
    }

    return 1;
}

int main() {
    int i, j, start;

    printf("Enter number of vertices: ");
    scanf("%d", &n);

    if (n <= 0 || n > 20) return 0;

    printf("Enter adjacency matrix:\n");
    for (i = 0; i < n; i++)
        for (j = 0; j < n; j++)
            scanf("%d", &a[i][j]);

    for (i = 0; i < n; i++)
        visited[i] = 0;

    printf("Enter starting vertex: ");
    scanf("%d", &start);

    if (start < 0 || start >= n) return 0;

    printf("DFS Traversal: ");
    dfs(start);

    for (i = 0; i < n; i++)
        if (visited[i] == 0)
            dfs(i);

    return 0;
}

5. Traverse an Adjacency Matrix Using Breadth First Search (BFS)


#include <stdio.h>

int a[20][20], visited[20], queue[20], n;

int bfs(int start) {
    int front = 0, rear = 0, v, i;

    visited[start] = 1;
    queue[rear++] = start;

    while (front < rear) {
        v = queue[front++];
        printf("%d ", v);

        for (i = 0; i < n; i++) {
            if (a[v][i] == 1 && visited[i] == 0) {
                visited[i] = 1;
                queue[rear++] = i;
            }
        }
    }

    return 1;
}

int main() {

        visited[i] = 0;

    printf("Enter starting vertex: ");
    scanf("%d", &start);

    printf("BFS Traversal: ");

    int vertex;
    struct node *next;
};

struct node *adj[20];
int visited[20], queue[20], n;

    struct node *p = malloc(sizeof(struct node));

    if (p == NULL) exit(1);

    p->next = adj[v];

    return 1;
}

int bfs(int start) {
    int front = 0, rear = 0, v;
    struct node *p;

    queue[rear++] = start;
