Question 1

Enter a matrix of 4 rows and 2 columns from the user and display it in matrix format.

Answer

#include <stdio.h>

int main() {
    int a[4][2];

    printf("Enter 8 elements:\n");

    for (int i = 0; i < 4; i++) {
        for (int j = 0; j < 2; j++) {
            scanf("%d", &a[i][j]);
        }
    }

    printf("Matrix:\n");

    for (int i = 0; i < 4; i++) {
        for (int j = 0; j < 2; j++) {
            printf("%d\t", a[i][j]);
        }
        printf("\n");
    }

    return 0;
}

Question 4

Enter two matrices from the user, multiply the matrices and display the answer.

Answer

#include <stdio.h>

int main() {
    int a[2][2], b[2][2], c[2][2];

    printf("Enter first matrix:\n");
    for (int i = 0; i < 2; i++) {
        for (int j = 0; j < 2; j++) {
            scanf("%d", &a[i][j]);
        }
    }

    printf("Enter second matrix:\n");
    for (int i = 0; i < 2; i++) {
        for (int j = 0; j < 2; j++) {
            scanf("%d", &b[i][j]);
        }
    }

    for (int i = 0; i < 2; i++) {
        for (int j = 0; j < 2; j++) {
            c[i][j] = 0;
            for (int k = 0; k < 2; k++) {
                c[i][j] += a[i][k] * b[k][j];
            }
        }
    }

    printf("Product of matrices:\n");
    for (int i = 0; i < 2; i++) {
        for (int j = 0; j < 2; j++) {
            printf("%d\t", c[i][j]);
        }
        printf("\n");
    }

    return 0;
}

Question 7

Create a Tic Tac Toe game using 2D arrays.

Answer

#include <stdio.h>

int main() {
    char board[3][3] = {
        {'1', '2', '3'},
        {'4', '5', '6'},
        {'7', '8', '9'}
    };

    int player = 1, choice, row, col, moves = 0;
    char mark;
    int winner = 0;

    while (moves < 9 && winner == 0) {
        printf("\n");

        for (int i = 0; i < 3; i++) {
            printf(" ");
            for (int j = 0; j < 3; j++) {
                printf(" %c ", board[i][j]);
                if (j < 2) printf("|");
            }
            printf("\n");
            if (i < 2) printf("---+---+---\n");
        }

        mark = (player == 1) ? 'X' : 'O';

        printf("Player %d (%c), enter position (1-9): ",
               player, mark);

        if (scanf("%d", &choice) != 1) {
            return 0;
        }

        if (choice < 1 || choice > 9) {
            printf("Invalid position. Try again.\n");
            continue;
        }

        row = (choice - 1) / 3;
        col = (choice - 1) % 3;

        if (board[row][col] == 'X' ||
            board[row][col] == 'O') {
            printf("Position already occupied. Try again.\n");
            continue;
        }

        board[row][col] = mark;
        moves++;

        for (int i = 0; i < 3; i++) {
            if (board[i][0] == mark &&
                board[i][1] == mark &&
                board[i][2] == mark)
                winner = 1;

            if (board[0][i] == mark &&
                board[1][i] == mark &&
                board[2][i] == mark)
                winner = 1;
        }

        if (board[0][0] == mark &&
            board[1][1] == mark &&
            board[2][2] == mark)
            winner = 1;

        if (board[0][2] == mark &&
            board[1][1] == mark &&
            board[2][0] == mark)
            winner = 1;

        if (winner == 0)
            player = (player == 1) ? 2 : 1;
    }

    printf("\nFinal board:\n");

    for (int i = 0; i < 3; i++) {
        printf(" ");
        for (int j = 0; j < 3; j++) {
            printf(" %c ", board[i][j]);
            if (j < 2) printf("|");
        }
        printf("\n");
        if (i < 2) printf("---+---+---\n");
    }

    if (winner)
        printf("Player %d (%c) wins!\n", player, mark);
    else
        printf("It's a draw!\n");

    return 0;
}
