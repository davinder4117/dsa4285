Question 1

Construct binary tree from preorder and inorder traversal.

Answer 1

#include <stdio.h>
#include <stdlib.h>

struct Node
{
    int data;
    struct Node *left;
    struct Node *right;
};

struct Node *createNode(int value)
{
    struct Node *temp = malloc(sizeof(struct Node));

    if (temp == NULL)
    {
        printf("Memory allocation failed\n");
        exit(1);
    }

    temp->data = value;
    temp->left = NULL;
    temp->right = NULL;

    return temp;
}

int search(int inorder[], int start, int end, int value)
{
    int i;

    for (i = start; i <= end; i++)
        if (inorder[i] == value)
            return i;

    return -1;
}

struct Node *buildTree(int preorder[], int inorder[],
                       int *preIndex, int start, int end)
{
    int value, pos;
    struct Node *root;

    if (start > end)
        return NULL;

    value = preorder[*preIndex];
    (*preIndex)++;

    root = createNode(value);
    pos = search(inorder, start, end, value);

    if (pos == -1)
    {
        printf("Invalid traversals\n");
        exit(1);
    }

    root->left = buildTree(preorder, inorder, preIndex,
                           start, pos - 1);
    root->right = buildTree(preorder, inorder, preIndex,
                            pos + 1, end);

    return root;
}

void postorder(struct Node *root)
{
    if (root != NULL)
    {
        postorder(root->left);
        postorder(root->right);
        printf("%d ", root->data);
    }
}

int main()
{
    int preorder[] = {1, 2, 4, 3, 5};
    int inorder[] = {4, 2, 1, 3, 5};
    int preIndex = 0;
    int n = 5;

    struct Node *root = buildTree(preorder, inorder,
                                  &preIndex, 0, n - 1);

    printf("Postorder traversal: ");
    postorder(root);

    return 0;
}

Question 2

Perform binary tree level order traversal.

Answer 2

#include <stdio.h>
#include <stdlib.h>

struct Node
{
    int data;
    struct Node *left;
    struct Node *right;
};

struct Node *createNode(int value)
{
    struct Node *temp = malloc(sizeof(struct Node));

    if (temp == NULL)
        exit(1);

    temp->data = value;
    temp->left = NULL;
    temp->right = NULL;

    return temp;
}

void levelOrder(struct Node *root)
{
    struct Node *queue[100];
    int front = 0, rear = 0;

    if (root == NULL)
        return;

    queue[rear++] = root;

    while (front < rear)
    {
        struct Node *temp = queue[front++];

        printf("%d ", temp->data);

        if (temp->left != NULL)
            queue[rear++] = temp->left;

        if (temp->right != NULL)
            queue[rear++] = temp->right;
    }
}

int main()
{
    struct Node *root = createNode(1);

    root->left = createNode(2);
    root->right = createNode(3);
    root->left->left = createNode(4);
    root->left->right = createNode(5);

    printf("Level order traversal: ");
    levelOrder(root);

    return 0;
}

Question 6

Find the maximum depth of a binary tree.

Answer 6

#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *left, *right;
};

struct Node *createNode(int data) {
    struct Node *newNode = malloc(sizeof(struct Node));
    newNode->data = data;
    newNode->left = NULL;
    newNode->right = NULL;
    return newNode;
}

int maxDepth(struct Node *root) {
    if (root == NULL)
        return 0;

    int leftDepth = maxDepth(root->left);
    int rightDepth = maxDepth(root->right);

    return 1 + (leftDepth > rightDepth ? leftDepth : rightDepth);
}

int main() {
    struct Node *root = createNode(1);
    root->left = createNode(2);
    root->right = createNode(3);
    root->left->left = createNode(4);
    root->left->right = createNode(5);

    printf("Maximum depth = %d\n", maxDepth(root));

    return 0;
}

Question 7

Create a Binary Search Tree using arrays and perform insertion, deletion, searching, preorder, inorder, postorder traversal, and finding the minimum element.

Answer 7

#include <stdio.h>

#define SIZE 100

int tree[SIZE];
int occupied[SIZE] = {0};

void insert(int index, int value) {
    if (index >= SIZE) {
        printf("Tree is full.\n");
        return;
    }

    if (!occupied[index]) {
        tree[index] = value;
        occupied[index] = 1;
        return;
    }

    if (value < tree[index])
        insert(index * 2, value);
    else if (value > tree[index])
        insert(index * 2 + 1, value);
    else
        printf("Duplicate value.\n");
}

int search(int index, int value) {
    if (index >= SIZE || !occupied[index])
        return 0;

    if (tree[index] == value)
        return 1;

    if (value < tree[index])
        return search(index * 2, value);

    return search(index * 2 + 1, value);
}

void preorder(int index) {
    if (index >= SIZE || !occupied[index])
        return;

    printf("%d ", tree[index]);
    preorder(index * 2);
    preorder(index * 2 + 1);
}

void inorder(int index) {
    if (index >= SIZE || !occupied[index])
        return;

    inorder(index * 2);
    printf("%d ", tree[index]);
    inorder(index * 2 + 1);
}

void postorder(int index) {
    if (index >= SIZE || !occupied[index])
        return;

    postorder(index * 2);
    postorder(index * 2 + 1);
    printf("%d ", tree[index]);
}

int minimum(int index) {
    while (index < SIZE && occupied[index]) {
        if (index * 2 >= SIZE || !occupied[index * 2])
            return tree[index];

        index = index * 2;
    }

    return -1;
}

void deleteValue(int index, int value) {
    if (index >= SIZE || !occupied[index])
        return;

    if (value < tree[index]) {
        deleteValue(index * 2, value);
    } else if (value > tree[index]) {
        deleteValue(index * 2 + 1, value);
    } else {
        int left = index * 2;
        int right = index * 2 + 1;

        if ((left >= SIZE || !occupied[left]) &&
            (right >= SIZE || !occupied[right])) {
            occupied[index] = 0;
        } else if (right < SIZE && occupied[right]) {
            int successorIndex = right;

            while (successorIndex * 2 < SIZE &&
                   occupied[successorIndex * 2])
                successorIndex *= 2;

            tree[index] = tree[successorIndex];
            deleteValue(right, tree[successorIndex]);
        } else {
            int predecessorIndex = left;

            while (predecessorIndex * 2 + 1 < SIZE &&
                   occupied[predecessorIndex * 2 + 1])
                predecessorIndex = predecessorIndex * 2 + 1;

            tree[index] = tree[predecessorIndex];
            deleteValue(left, tree[predecessorIndex]);
        }
    }
}

int main() {
    int choice, value;

    do {
        printf("\n1.Insert 2.Delete 3.Search");
        printf("\n4.Preorder 5.Inorder 6.Postorder");
        printf("\n7.Minimum 8.Exit\n");
        scanf("%d", &choice);

        switch (choice) {
            case 1:
                printf("Enter value: ");
                scanf("%d", &value);
                insert(1, value);
                break;

            case 2:
                printf("Enter value to delete: ");
                scanf("%d", &value);
                deleteValue(1, value);
                break;

            case 3:
                printf("Enter value to search: ");
                scanf("%d", &value);
                if (search(1, value))
                    printf("Value found.\n");
                else
                    printf("Value not found.\n");
                break;

            case 4:
                preorder(1);
                printf("\n");
                break;

            case 5:
                inorder(1);
                printf("\n");
                break;

            case 6:
                postorder(1);
                printf("\n");
                break;

            case 7:
                if (occupied[1])
                    printf("Minimum = %d\n", minimum(1));
                else
                    printf("Tree is empty.\n");
                break;
        }
    } while (choice != 8);

    return 0;
}

Note: This array representation uses index 1 for the root, 2 × index for the left child, and 2 × index + 1 for the right child. Because of this representation, a tree with very uneven branches can run out of array positions.

Create a Binary Search Tree dynamically and perform insertion, deletion, searching, preorder, inorder, postorder traversal, and finding the minimum element.

Answer 8

#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *left, *right;
};

struct Node *createNode(int value) {
    struct Node *newNode = malloc(sizeof(struct Node));

    if (newNode == NULL) {
        printf("Memory allocation failed.\n");
        exit(1);
    }

    newNode->data = value;
    newNode->left = NULL;
    newNode->right = NULL;
    return newNode;
}

struct Node *insert(struct Node *root, int value) {
    if (root == NULL)
        return createNode(value);

    if (value < root->data)
        root->left = insert(root->left, value);
    else if (value > root->data)
        root->right = insert(root->right, value);
    else
        printf("Duplicate value.\n");

    return root;
}

struct Node *minimum(struct Node *root) {
    while (root != NULL && root->left != NULL)
        root = root->left;

    return root;
}

struct Node *deleteNode(struct Node *root, int value) {
    if (root == NULL)
        return NULL;

    if (value < root->data) {
        root->left = deleteNode(root->left, value);
    } else if (value > root->data) {
        root->right = deleteNode(root->right, value);
    } else {
        if (root->left == NULL) {
            struct Node *temp = root->right;
            free(root);
            return temp;
        }

        if (root->right == NULL) {
            struct Node *temp = root->left;
            free(root);
            return temp;
        }

        struct Node *temp = minimum(root->right);
        root->data = temp->data;
        root->right = deleteNode(root->right, temp->data);
    }

    return root;
}

struct Node *search(struct Node *root, int value) {
    if (root == NULL || root->data == value)
        return root;

    if (value < root->data)
        return search(root->left, value);

    return search(root->right, value);
}

void preorder(struct Node *root) {
    if (root != NULL) {
        printf("%d ", root->data);
        preorder(root->left);
        preorder(root->right);
    }
}

void inorder(struct Node *root) {
    if (root != NULL) {
        inorder(root->left);
        printf("%d ", root->data);
        inorder(root->right);
    }
}

void postorder(struct Node *root) {
    if (root != NULL) {
        postorder(root->left);
        postorder(root->right);
        printf("%d ", root->data);
    }
}

int main() {
    struct Node *root = NULL;
    struct Node *found;
    int choice, value;

    do {
        printf("\n1.Insert 2.Delete 3.Search");
        printf("\n4.Preorder 5.Inorder 6.Postorder");
        printf("\n7.Minimum 8.Exit\n");
        scanf("%d", &choice);

        switch (choice) {
            case 1:
                printf("Enter value: ");
                scanf("%d", &value);
                root = insert(root, value);
                break;

            case 2:
                printf("Enter value to delete: ");
                scanf("%d", &value);
                root = deleteNode(root, value);
                break;

            case 3:
                printf("Enter value to search: ");
                scanf("%d", &value);
                found = search(root, value);
                if (found != NULL)
                    printf("Value found.\n");
                else
                    printf("Value not found.\n");
                break;

            case 4:
                preorder(root);
                printf("\n");
                break;

            case 5:
                inorder(root);
                printf("\n");
                break;

            case 6:
                postorder(root);
                printf("\n");
                break;

            case 7:
                found = minimum(root);
                if (found != NULL)
                    printf("Minimum = %d\n", found->data);
                else
                    printf("Tree is empty.\n");
                break;
        }
    } while (choice != 8);

    return 0;
}

Question 9

Find the inorder predecessor and successor of a given value in a dynamically created Binary Search Tree.

Answer 9

#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *left, *right;
};

struct Node *createNode(int value) {
    struct Node *node = malloc(sizeof(struct Node));
    node->data = value;
    node->left = NULL;
    node->right = NULL;
    return node;
}

struct Node *insert(struct Node *root, int value) {
    if (root == NULL)
        return createNode(value);

    if (value < root->data)
        root->left = insert(root->left, value);
    else if (value > root->data)
        root->right = insert(root->right, value);

    return root;
}

void findPredecessorSuccessor(
    struct Node *root, int key,
    struct Node **predecessor,
    struct Node **successor) {

    *predecessor = NULL;
    *successor = NULL;

    struct Node *current = root;

    while (current != NULL) {
        if (current->data == key) {
            if (current->left != NULL) {
                struct Node *temp = current->left;
                while (temp->right != NULL)
                    temp = temp->right;
                *predecessor = temp;
            }

            if (current->right != NULL) {
                struct Node *temp = current->right;
                while (temp->left != NULL)
                    temp = temp->left;
                *successor = temp;
            }

            return;
        } else if (key < current->data) {
            *successor = current;
            current = current->left;
        } else {
            *predecessor = current;
            current = current->right;
        }
    }
}

int main() {
    struct Node *root = NULL;
    struct Node *predecessor, *successor;
    int n, value, key, i;

    printf("Enter number of nodes: ");
    scanf("%d", &n);

    for (i = 0; i < n; i++) {
        printf("Enter value: ");
        scanf("%d", &value);
        root = insert(root, value);
    }

    printf("Enter value to find predecessor and successor: ");
    scanf("%d", &key);

    findPredecessorSuccessor(root, key, &predecessor, &successor);

    if (predecessor != NULL)
        printf("Predecessor = %d\n", predecessor->data);
    else
        printf("Predecessor does not exist.\n");

    if (successor != NULL)
        printf("Successor = %d\n", successor->data);
    else
        printf("Successor does not exist.\n");

    return 0;
}

Question 10

Check whether a Binary Search Tree contains a dead end.

Answer 10

#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *left, *right;
};

struct Node *createNode(int value) {
    struct Node *node = malloc(sizeof(struct Node));
    node->data = value;
    node->left = NULL;
    node->right = NULL;
    return node;
}

struct Node *insert(struct Node *root, int value) {
    if (root == NULL)
        return createNode(value);

    if (value < root->data)
        root->left = insert(root->left, value);
    else if (value > root->data)
        root->right = insert(root->right, value);

    return root;
}

int hasDeadEnd(struct Node *root, int minValue, int maxValue) {
    if (root == NULL)
        return 0;

    if (minValue == maxValue)
        return 1;

    return hasDeadEnd(root->left, minValue, root->data - 1) ||
           hasDeadEnd(root->right, root->data + 1, maxValue);
}

int main() {
    struct Node *root = NULL;
    int n, value, i;

    printf("Enter number of nodes: ");
    scanf("%d", &n);

    printf("Enter positive integer values:\n");

    for (i = 0; i < n; i++) {
        scanf("%d", &value);
        root = insert(root, value);
    }

    if (hasDeadEnd(root, 1, 1000000))
        printf("BST contains a dead end.\n");
    else
        printf("BST does not contain a dead end.\n");

    return 0;
}

int compare(const void *x, const void *y) {
    int p = *(int *)x, q = *(int *)y;
    return (p > q) - (p < q);
}

int convert(int i) {
    if (i >= n) return 0;
    convert(2 * i + 1);
    a[i] = sorted[k++];
    convert(2 * i + 2);
    return 1;

15. Construct BST from Preorder Traversal Using Array


#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

int a[100], n, pos = 0;

struct node {
    int data;
    struct node *left, *right;
};

struct node *build(int low, int high) {
    struct node *root;

    if (pos >= n || a[pos] <= low || a[pos] >= high)
        return NULL;

    root = malloc(sizeof(struct node));
    if (root == NULL) exit(1);

    root->data = a[pos++];
    root->left = build(low, root->data);
    root->right = build(root->data, high);

    return root;
}

int inorder(struct node *root) {
    if (root == NULL) return 0;

    inorder(root->left);

16. Construct BST from Postorder Traversal Using Array


#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

int a[100], n, pos;

struct node {
    int data;
    struct node *left, *right;
};

struct node *build(int low, int high) {
    struct node *root;

    if (pos < 0 || a[pos] <= low || a[pos] >= high)
        return NULL;

    root = malloc(sizeof(struct node));
    if (root == NULL) exit(1);

    root->data = a[pos--];
    root->right = build(root->data, high);
    root->left = build(low, root->data);

    return root;
}

int inorder(struct node *root) {
    if (root == NULL) return 0;

    inorder(root->left);
    printf("%d ", root->data);
    inorder(root->right);

    return 1;
}

int main() {
    int i;
    struct node *root;

    printf("Enter number of nodes: ");
    scanf("%d", &n);

    if (n <= 0 || n > 100) return 0;

    printf("Enter postorder traversal: ");
    for (i = 0; i < n; i++) scanf("%d", &a[i]);

    pos = n - 1;
    root = build(INT_MIN, INT_MAX);

    if (pos != -1) {
        printf("Invalid postorder traversal");
        return 0;
    }

    printf("Inorder traversal: ");
    inorder(root);

    return 0;
}

18. Create Almost Complete Binary Tree Dynamically


#include <stdio.h>
#include <stdlib.h>

    struct node *left, *right;
};

    struct node *p = malloc(sizeof(struct node));

    if (p == NULL) exit(1);

    p->data = value;
    p->left = NULL;
    p->right = NULL;

    return p;
}

int levelOrder(struct node *root) {
    struct node *queue[1000], *temp;
    int front = 0, rear = 0;

    if (root == NULL) return 0;

    queue[rear++] = root;

    while (front < rear) {
        temp = queue[front++];
        printf("%d ", temp->data);

        if (temp->left != NULL) queue[rear++] = temp->left;
        if (temp->right != NULL) queue[rear++] = temp->right;
    }

    return 1;
}

int main() {
    struct node *root = NULL, *queue[1000], *temp;
    int n, i, value, front = 0, rear = 0;

    printf("Enter number of nodes (maximum 1000): ");
    scanf("%d", &n);

    if (n <= 0 || n > 1000) return 0;

    for (i = 0; i < n; i++) {
        printf("Enter value: ");
        scanf("%d", &value);

        if (root == NULL) {
            root = createNode(value);
            queue[rear++] = root;
        } else {
            temp = queue[front];

            if (temp->left == NULL) {

20. Count Nodes in a Dynamically Constructed BST


#include <stdio.h>
#include <stdlib.h>

struct node {
    int data;
    struct node *left, *right;
};

struct node *insert(struct node *root, int value) {
    if (root == NULL) {
        struct node *p = malloc(sizeof(struct node));

        if (p == NULL) exit(1);

        p->data = value;
        p->left = NULL;
        p->right = NULL;

        return p;
    }

    if (value < root->data)
        root->left = insert(root->left, value);
    else if (value > root->data)
        root->right = insert(root->right, value);

    return root;
}

int countNodes(struct node *root) {
    if (root == NULL) return 0;

    return 1 + countNodes(root->left)
             + countNodes(root->right);
}

int inorder(struct node *root) {
    if (root == NULL) return 0;

    inorder(root->left);
    printf("%d ", root->data);
    inorder(root->right);

    return 1;
}

int main() {
    struct node *root = NULL;
    int n, i, value;

    printf("Enter number of nodes: ");
    scanf("%d", &n);

    if (n < 0) return 0;

    for (i = 0; i < n; i++) {
        printf("Enter value: ");
        scanf("%d", &value);
        root = insert(root, value);
    }

    printf("Inorder traversal: ");
    inorder(root);

    printf("\nTotal number of nodes = %d", countNodes(root));

    return 0;
}
