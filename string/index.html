Question 1

Enter the string from user and convert it into uppercase.

Answer

Question 20

Enter the string and count the number of characters, words, spaces and lines.

Answer

#include <stdio.h>
#include <ctype.h>
#include <string.h>
int main() {
    char s[1000];
    int characters = 0, words = 0;
    int spaces = 0, lines = 0, inWord = 0;

    if (fgets(s, sizeof(s), stdin) != NULL) {
        lines = 1;

        for (int i = 0; s[i] != '\0'; i++) {
            if (s[i] == '\n') {
                lines++;
                continue;
            }

            characters++;

            if (s[i] == ' ') {
                spaces++;
            }

            if (isspace((unsigned char)s[i])) {
                inWord = 0;
            } else if (!inWord) {
                words++;
                inWord = 1;
            }
        }

        if (strchr(s, '\n') != NULL) {
            lines--;
        }
    }

    printf("Characters = %d\n", characters);
    printf("Words = %d\n", words);
    printf("Spaces = %d\n", spaces);
    printf("Lines = %d\n", lines);

    return 0;
}

Question 21

Print all permutations of the string ABCDE.

Answer

#include <stdio.h>
#include <string.h>

void swap(char *a, char *b) {
    char temp = *a;
    *a = *b;
    *b = temp;
}

void permute(char s[], int start, int end) {
    if (start == end) {
        printf("%s\n", s);
        return;
    }

    for (int i = start; i <= end; i++) {
        swap(&s[start], &s[i]);
        permute(s, start + 1, end);
        swap(&s[start], &s[i]);
    }
}

int main() {
    char s[] = "ABCDE";
    permute(s, 0, strlen(s) - 1);
    return 0;
}

Question 22

    for (int i = 0; i < n - 1; i++) {
        for (int j = 0; j < n - 1 - i; j++) {
            if ((unsigned char)s[j] >
                (unsigned char)s[j + 1]) {

        }

}

Question 23

Enter a sentence and a word, then count the occurrences of that word in the sentence.

Answer

#include <stdio.h>
#include <string.h>
#include <ctype.h>
int main() {
    char sentence[1000], word[100];
    int count = 0;

    fgets(sentence, sizeof(sentence), stdin);
    fgets(word, sizeof(word), stdin);

    sentence[strcspn(sentence, "\n")] = '\0';
    word[strcspn(word, "\n")] = '\0';

    if (word[0] == '\0') {
        printf("Word cannot be empty");
        return 0;
    }

    char *p = sentence;
    int len = strlen(word);

    while (*p != '\0') {
        while (*p != '\0' &&
               isspace((unsigned char)*p)) {
            p++;
        }

        if (*p == '\0') {
            break;
        }

        char *start = p;

        while (*p != '\0' &&
               !isspace((unsigned char)*p)) {
            p++;
        }

        if (p - start == len &&
            strncmp(start, word, len) == 0) {
            count++;
        }
    }

    printf("Occurrences = %d", count);
    return 0;
}

Question 24

Enter a sentence, a word to find and another word to replace it with.

Answer

#include <stdio.h>
#include <string.h>
#include <ctype.h>
int main() {
    char sentence[1000], word[100], replacement[100];
    char result[2000] = "";
    int i = 0;

    fgets(sentence, sizeof(sentence), stdin);
    fgets(word, sizeof(word), stdin);
    fgets(replacement, sizeof(replacement), stdin);

    sentence[strcspn(sentence, "\n")] = '\0';
    word[strcspn(word, "\n")] = '\0';
    replacement[strcspn(replacement, "\n")] = '\0';

    if (word[0] == '\0') {
        printf("Word cannot be empty");
        return 0;
    }

    while (sentence[i] != '\0') {
        if (isspace((unsigned char)sentence[i])) {
            int len = strlen(result);
            result[len] = sentence[i];
            result[len + 1] = '\0';
            i++;
            continue;
        }

        int start = i;

        while (sentence[i] != '\0' &&
               !isspace((unsigned char)sentence[i])) {
            i++;
        }

        int len = i - start;

        if (len == strlen(word) &&
            strncmp(&sentence[start], word, len) == 0) {
            strcat(result, replacement);
        } else {
            int current = strlen(result);

            strncat(result, &sentence[start], len);
            result[current + len] = '\0';
        }
    }

    printf("%s", result);
    return 0;
}

Question 25

Enter the string and trim spaces from both ends.

Answer

#include <stdio.h>
#include <string.h>
#include <ctype.h>
int main() {
    char s[200];
    int start = 0, end, j = 0;

    fgets(s, sizeof(s), stdin);
    s[strcspn(s, "\n")] = '\0';

    end = strlen(s) - 1;

    while (s[start] != '\0' &&
           isspace((unsigned char)s[start])) {
        start++;
    }

    while (end >= start &&
           isspace((unsigned char)s[end])) {
        end--;
    }

    for (int i = start; i <= end; i++) {
        s[j++] = s[i];
    }

    s[j] = '\0';

    printf("%s", s);
    return 0;
}

Question 27

Display the string with the user's chosen alignment: 1 for left, 2 for center and 3 for right.

Answer

#include <stdio.h>
#include <string.h>
int main() {
    char s[200];
    int option, width = 40, len, spaces;

    fgets(s, sizeof(s), stdin);
    s[strcspn(s, "\n")] = '\0';

    scanf("%d", &option);
    len = strlen(s);

    if (len > width) {
        width = len;
    }

    if (option == 1) {
        printf("%s", s);
    } else if (option == 2) {
        spaces = (width - len) / 2;

        for (int i = 0; i < spaces; i++) {
            printf(" ");
        }

        printf("%s", s);
    } else if (option == 3) {
        spaces = width - len;

        for (int i = 0; i < spaces; i++) {
            printf(" ");
        }

        printf("%s", s);
    } else {
        printf("Invalid option");
    }

    return 0;
}

Question 28

Create a dictionary of 10 words and meanings with create, delete and search operations.

Answer

#include <stdio.h>
#include <string.h>

#define SIZE 10

struct Entry {
    char word[50];
    char meaning[100];
    int active;
};

int main() {
    struct Entry d[SIZE] = {0};
    int choice, i, found;
    char word[50];

    while (1) {
        printf("\n1. Create\n2. Delete\n3. Search\n4. Exit\n");
        scanf("%d", &choice);

        if (choice == 1) {
            found = -1;

            for (i = 0; i < SIZE; i++) {
                if (!d[i].active) {
                    found = i;
                    break;
                }
            }

            if (found == -1) {
                printf("Dictionary is full\n");
                continue;
            }

            printf("Enter word: ");
            scanf("%49s", d[found].word);

            printf("Enter meaning: ");
            scanf(" %99[^\n]", d[found].meaning);

            d[found].active = 1;

        } else if (choice == 2) {
            printf("Enter word to delete: ");
            scanf("%49s", word);
            found = 0;

            for (i = 0; i < SIZE; i++) {
                if (d[i].active &&
                    strcmp(d[i].word, word) == 0) {
                    d[i].active = 0;
                    found = 1;
                    break;
                }
            }

            if (found) {
                printf("Deleted\n");
            } else {
                printf("Word not found\n");
            }

        } else if (choice == 3) {
            printf("Enter word to search: ");
            scanf("%49s", word);
            found = 0;

            for (i = 0; i < SIZE; i++) {
                if (d[i].active &&
                    strcmp(d[i].word, word) == 0) {
                    printf("%s: %s\n",
                           d[i].word, d[i].meaning);
                    found = 1;
                    break;
                }
            }

            if (!found) {
                printf("Word not found\n");
            }

        } else if (choice == 4) {
            break;
        } else {
            printf("Invalid choice\n");
        }
    }

    return 0;
}
