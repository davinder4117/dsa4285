Question 1

Calculate factorial of a number using recursion.

Answer

#include <stdio.h>

int factorial(int n) {
    if (n <= 1)
        return 1;
    return n * factorial(n - 1);
}

int main() {
    int n;
    scanf("%d", &n);
    printf("%d", factorial(n));
    return 0;
}

Question 2

Calculate Fibonacci of a number using recursion.

Answer

#include <stdio.h>

int fibonacci(int n) {
    if (n <= 1)
        return n;
    return fibonacci(n - 1) + fibonacci(n - 2);
}

int main() {
    int n;
    scanf("%d", &n);

    for (int i = 0; i < n; i++)
        printf("%d ", fibonacci(i));

    return 0;
}

Question 3

Calculate the sum of natural numbers till 10 using recursion.

Answer

#include <stdio.h>

int sum(int n) {
    if (n == 0)
        return 0;
    return n + sum(n - 1);
}

int main() {
    printf("%d", sum(10));
    return 0;
}

Question 4

Convert a number into binary using recursion.

Answer

#include <stdio.h>

void binary(int n) {
    if (n > 1)
        binary(n / 2);
    printf("%d", n % 2);
}

int main() {
    int n;
    scanf("%d", &n);

    if (n < 0) {
        printf("Enter a non-negative number");
        return 0;
    }

    binary(n);
    return 0;
}

Question 5

Find the largest element in the integer array using recursion.

Answer

#include <stdio.h>

int largest(int a[], int n) {
    if (n == 1)
        return a[0];

    int max = largest(a, n - 1);

    if (a[n - 1] > max)
        return a[n - 1];

    return max;
}

int main() {
    int a[100], n;

    scanf("%d", &n);

    if (n <= 0 || n > 100) {
        printf("Invalid array size");
        return 0;
    }

    for (int i = 0; i < n; i++)
        scanf("%d", &a[i]);

    printf("%d", largest(a, n));
    return 0;
}

Question 7

Enter a number from the user and add all its digits using recursion.

Answer

#include <stdio.h>

int sumDigits(int n) {
    if (n == 0)
        return 0;

    return n % 10 + sumDigits(n / 10);
}

int main() {
    int n;
    scanf("%d", &n);

    if (n < 0)
        n = -n;

    printf("%d", sumDigits(n));
    return 0;
}

Question 8

Find the arithmetic mean of an integer array using recursion.

Answer

#include <stdio.h>

int sumArray(int a[], int n) {
    if (n == 0)
        return 0;

    return a[n - 1] + sumArray(a, n - 1);
}

int main() {
    int a[100], n;

    scanf("%d", &n);

    if (n <= 0 || n > 100) {
        printf("Invalid array size");
        return 0;
    }

    for (int i = 0; i < n; i++)
        scanf("%d", &a[i]);

    double mean = (double)sumArray(a, n) / n;

    printf("%.2f", mean);
    return 0;
}
