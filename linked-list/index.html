Question 1

Implement the operations of a singly linked list storing integer values (Insert at Begin, Insert at Position, Insert at End, Delete at Begin, Delete at End, Delete at Position, IsEmpty, IsFull, Display, Search).

Answer 1

#include <stdio.h>
#include <stdlib.h>

struct Node
{
    int data;
    struct Node *next;
};

struct Node *start = NULL;

int isEmpty()
{
    return start == NULL;
}

int isFull()
{
    struct Node *temp = malloc(sizeof(struct Node));
    if (temp == NULL)
        return 1;
    free(temp);
    return 0;
}

void insertBegin(int value)
{
    struct Node *temp = malloc(sizeof(struct Node));
    if (temp == NULL)
    {
        printf("Memory full\n");
        return;
    }

    temp->data = value;
    temp->next = start;
    start = temp;
}

void insertEnd(int value)
{
    struct Node *temp = malloc(sizeof(struct Node));
    struct Node *last = start;

    if (temp == NULL)
    {
        printf("Memory full\n");
        return;
    }

    temp->data = value;
    temp->next = NULL;

    if (start == NULL)
        start = temp;
    else
    {
        while (last->next != NULL)
            last = last->next;
        last->next = temp;
    }
}

void insertPos(int value, int pos)
{
    struct Node *temp, *last = start;
    int i;

    if (pos < 1)
    {
        printf("Invalid position\n");
        return;
    }

    if (pos == 1)
    {
        insertBegin(value);
        return;
    }

    for (i = 1; last != NULL && i < pos - 1; i++)
        last = last->next;

    if (last == NULL)
    {
        printf("Invalid position\n");
        return;
    }

    temp = malloc(sizeof(struct Node));
    if (temp == NULL)
    {
        printf("Memory full\n");
        return;
    }

    temp->data = value;
    temp->next = last->next;
    last->next = temp;
}

void deleteBegin()
{
    struct Node *temp;

    if (isEmpty())
    {
        printf("List is empty\n");
        return;
    }

    temp = start;
    start = start->next;
    free(temp);
}

void deleteEnd()
{
    struct Node *last = start;

    if (isEmpty())
    {
        printf("List is empty\n");
        return;
    }

    if (start->next == NULL)
    {
        free(start);
        start = NULL;
        return;
    }

    while (last->next->next != NULL)
        last = last->next;

    free(last->next);
    last->next = NULL;
}

void deletePos(int pos)
{
    struct Node *last = start, *temp;
    int i;

    if (pos < 1 || isEmpty())
    {
        printf("Invalid position or empty list\n");
        return;
    }

    if (pos == 1)
    {
        deleteBegin();
        return;
    }

    for (i = 1; last != NULL && i < pos - 1; i++)
        last = last->next;

    if (last == NULL || last->next == NULL)
    {
        printf("Invalid position\n");
        return;
    }

    temp = last->next;
    last->next = temp->next;
    free(temp);
}

void display()
{
    struct Node *temp = start;

    if (isEmpty())
    {
        printf("List is empty\n");
        return;
    }

    while (temp != NULL)
    {
        printf("%d ", temp->data);
        temp = temp->next;
    }
    printf("\n");
}

void search(int value)
{
    struct Node *temp = start;
    int pos = 1;

    while (temp != NULL)
    {
        if (temp->data == value)
        {
            printf("Found at position %d\n", pos);
            return;
        }
        temp = temp->next;
        pos++;
    }

    printf("Not found\n");
}

int main()
{
    insertBegin(20);
    insertBegin(10);
    insertEnd(40);
    insertPos(30, 3);

    display();
    search(30);

    deleteBegin();
    deleteEnd();
    deletePos(1);

    display();

    printf("Empty: %d\n", isEmpty());
    printf("Full: %d\n", isFull());

    return 0;
}

#include <stdio.h>
#include <stdlib.h>

struct Node
{
    int coeff;
    int power;
    struct Node *next;
};

struct Node *createNode(int coeff, int power)
{
    struct Node *temp = malloc(sizeof(struct Node));

    if (temp == NULL)
    {
        printf("Memory allocation failed\n");
        exit(1);
    }

    temp->coeff = coeff;
    temp->power = power;
    temp->next = NULL;

    return temp;
}

void insertTerm(struct Node **start, int coeff, int power)
{
    struct Node *temp = createNode(coeff, power);
    struct Node *last;

    if (coeff == 0)
    {
        free(temp);
        return;
    }

    if (*start == NULL || (*start)->power < power)
    {
        temp->next = *start;
        *start = temp;
        return;
    }

    last = *start;

    while (last->next != NULL && last->next->power > power)
        last = last->next;

    if (last->power == power)
    {
        last->coeff += coeff;
        free(temp);
        return;
    }

    if (last->next != NULL && last->next->power == power)
    {
        last->next->coeff += coeff;
        free(temp);
        return;
    }

    temp->next = last->next;
    last->next = temp;
}

struct Node *readPolynomial()
{
    struct Node *poly = NULL;
    int n, coeff, power, i;

    printf("Enter number of terms: ");
    scanf("%d", &n);

    for (i = 0; i < n; i++)
    {
        printf("Enter coefficient and power: ");
        scanf("%d%d", &coeff, &power);
        insertTerm(&poly, coeff, power);
    }

    return poly;
}

struct Node *addPolynomial(struct Node *p, struct Node *q)
{
    struct Node *result = NULL;

    while (p != NULL)
    {
        insertTerm(&result, p->coeff, p->power);
        p = p->next;
    }

    while (q != NULL)
    {
        insertTerm(&result, q->coeff, q->power);
        q = q->next;
    }
