Question 9

Enter an array from user and find the k-th smallest value.

Answer

#include <stdio.h>
int main() {
    int a[100], n, k, temp;

    scanf("%d", &n);

    for (int i = 0; i < n; i++) {
        scanf("%d", &a[i]);
    }

    scanf("%d", &k);

    if (k < 1 || k > n) {
        printf("Invalid k");
        return 0;
    }

    for (int i = 0; i < n - 1; i++) {
        for (int j = 0; j < n - 1 - i; j++) {
            if (a[j] > a[j + 1]) {
                temp = a[j];
                a[j] = a[j + 1];
                a[j + 1] = temp;
            }
        }
    }

    printf("K-th smallest = %d", a[k - 1]);

    return 0;
}

Question 11

Enter an array from user and find the duplicate values.

Answer

#include <stdio.h>
int main() {
    int a[100], n, duplicate;

    scanf("%d", &n);

    for (int i = 0; i < n; i++) {
        scanf("%d", &a[i]);
    }

    for (int i = 0; i < n; i++) {
        duplicate = 0;

        for (int j = 0; j < i; j++) {
            if (a[i] == a[j]) {
                duplicate = 1;
                break;
            }
        }

        if (duplicate == 1) {
            continue;
        }

        for (int j = i + 1; j < n; j++) {
            if (a[i] == a[j]) {
                printf("%d ", a[i]);
                break;
            }
        }
    }

    return 0;
}

Question 14

Enter an array from user and print all possible non-empty combinations of its elements.

Answer

#include <stdio.h>

void combinations(int a[], int n, int start,
                  int result[], int size) {
    if (size > 0) {
        for (int i = 0; i < size; i++) {
            printf("%d ", result[i]);
        }
        printf("\n");
    }

    for (int i = start; i < n; i++) {
        result[size] = a[i];
        combinations(a, n, i + 1, result, size + 1);
    }
}

int main() {
    int a[20], result[20], n;

    scanf("%d", &n);

    if (n < 1 || n > 20) {
        return 0;
    }

    for (int i = 0; i < n; i++) {
        scanf("%d", &a[i]);
    }

    combinations(a, n, 0, result, 0);

    return 0;
}

Question 15

Enter an array from user and print all palindrome numbers.

Answer

#include <stdio.h>
int main() {
    int a[100], n, num, temp, digit, reverse;

    scanf("%d", &n);

    for (int i = 0; i < n; i++) {
        scanf("%d", &a[i]);
    }

    for (int i = 0; i < n; i++) {
        num = a[i];

        if (num < 0) {
            continue;
        }

        temp = num;
        reverse = 0;

        while (temp > 0) {
            digit = temp % 10;
            reverse = reverse * 10 + digit;
            temp = temp / 10;
        }

        if (num == reverse) {
            printf("%d ", num);
        }
    }

    return 0;
}

Question 16

        scanf("%d", &a[i]);
    }

    scanf("%d", &m);

    for (int i = 0; i < m; i++) {
        scanf("%d", &b[i]);
    }

    if (n != m) {
        same = 0;
    } else {
        for (int i = 0; i < n; i++) {
            if (a[i] != b[i]) {
                same = 0;
                break;
            }
        }
    }

    if (same == 1) {
        printf("Arrays are equal");
    } else {
        printf("Arrays are not equal");
    }

    return 0;
}

Question 17

Merge two sorted arrays into one sorted array.

Answer

#include <stdio.h>
int main() {
    int a[100], b[100], c[200];
    int n, m, i = 0, j = 0, k = 0;

    scanf("%d", &n);

    for (int x = 0; x < n; x++) {
        scanf("%d", &a[x]);
    }

    scanf("%d", &m);

    for (int x = 0; x < m; x++) {
        scanf("%d", &b[x]);
    }

    while (i < n && j < m) {
        if (a[i] <= b[j]) {
            c[k] = a[i];
            i++;
        } else {
            c[k] = b[j];
            j++;
        }
        k++;
    }

    while (i < n) {
        c[k++] = a[i++];
    }

    while (j < m) {
        c[k++] = b[j++];
    }

    for (int x = 0; x < n + m; x++) {
        printf("%d ", c[x]);
    }

    return 0;
}

Question 20

Trapping Rain Water: Given an array representing bar heights, calculate the amount of water trapped after raining.

Answer

#include <stdio.h>
int main() {
    int a[100], n;
    int left = 0, right;
    int leftMax = 0, rightMax = 0, water = 0;

    scanf("%d", &n);

    for (int i = 0; i < n; i++) {
        scanf("%d", &a[i]);
    }

    right = n - 1;

    while (left < right) {
        if (a[left] < a[right]) {
            if (a[left] >= leftMax) {
                leftMax = a[left];
            } else {
                water += leftMax - a[left];
            }

            left++;
        } else {
            if (a[right] >= rightMax) {
                rightMax = a[right];
            } else {
                water += rightMax - a[right];
            }

            right--;
        }
    }

    printf("Trapped water = %d", water);

    return 0;
}
