1. Selection Sort on an Integer Array (Iterative)


#include <stdio.h>
int main() {
    int a[100], n, i, j, min, temp;
    scanf("%d", &n);
    for (i = 0; i < n; i++) scanf("%d", &a[i]);
    for (i = 0; i < n - 1; i++) {
        min = i;
        for (j = i + 1; j < n; j++)
            if (a[j] < a[min]) min = j;
        temp = a[i]; a[i] = a[min]; a[min] = temp;
    }
    for (i = 0; i < n; i++) printf("%d ", a[i]);
    return 0;
}

6. Merge Sort on an Integer Array (Recursive)


#include <stdio.h>
void merge(int a[], int l, int m, int r) {
    int t[100], i = l, j = m + 1, k = 0;
    while (i <= m && j <= r) {
        if (a[i] <= a[j]) t[k++] = a[i++];
        else t[k++] = a[j++];
    }
    while (i <= m) t[k++] = a[i++];
    while (j <= r) t[k++] = a[j++];
    for (i = l, k = 0; i <= r; i++, k++) a[i] = t[k];
}
void mergeSort(int a[], int l, int r) {
    int m;
    if (l < r) {
        m = (l + r) / 2;
        mergeSort(a, l, m);
        mergeSort(a, m + 1, r);
        merge(a, l, m, r);
    }
}
int main() {
    int a[100], n, i;
    scanf("%d", &n);
    for (i = 0; i < n; i++) scanf("%d", &a[i]);
    mergeSort(a, 0, n - 1);
    for (i = 0; i < n; i++) printf("%d ", a[i]);
    return 0;
}

7. Merge Sort on an Integer Array (Iterative)


#include <stdio.h>
void merge(int a[], int t[], int l, int m, int r) {
    int i = l, j = m + 1, k = l;
    while (i <= m && j <= r) {
        if (a[i] <= a[j]) t[k++] = a[i++];
        else t[k++] = a[j++];
    }
    while (i <= m) t[k++] = a[i++];
    while (j <= r) t[k++] = a[j++];
    for (i = l; i <= r; i++) a[i] = t[i];
}
int main() {
    int a[100], t[100], n, i, w, l, m, r;
    scanf("%d", &n);
    for (i = 0; i < n; i++) scanf("%d", &a[i]);
    for (w = 1; w < n; w *= 2) {
        for (l = 0; l < n; l += 2 * w) {
            m = l + w - 1;
            r = l + 2 * w - 1;
            if (m >= n) continue;
            if (r >= n) r = n - 1;
            merge(a, t, l, m, r);
        }
    }
    for (i = 0; i < n; i++) printf("%d ", a[i]);
    return 0;
}

8. Quick Sort on an Integer Array (Recursive)


#include <stdio.h>
int partition(int a[], int low, int high) {
    int pivot = a[high], i = low - 1, j, temp;
    for (j = low; j < high; j++) {
        if (a[j] <= pivot) {
            i++;
            temp = a[i]; a[i] = a[j]; a[j] = temp;
        }
    }
    temp = a[i + 1]; a[i + 1] = a[high]; a[high] = temp;
    return i + 1;
}
void quickSort(int a[], int low, int high) {
    int p;
    if (low < high) {
        p = partition(a, low, high);
        quickSort(a, low, p - 1);
        quickSort(a, p + 1, high);
    }
}
int main() {
    int a[100], n, i;
    scanf("%d", &n);
    for (i = 0; i < n; i++) scanf("%d", &a[i]);
    quickSort(a, 0, n - 1);
    for (i = 0; i < n; i++) printf("%d ", a[i]);
    return 0;
}

9. Quick Sort on an Integer Array (Iterative)


#include <stdio.h>
int partition(int a[], int low, int high) {
    int pivot = a[high], i = low - 1, j, temp;
    for (j = low; j < high; j++) {
        if (a[j] <= pivot) {
            i++;
            temp = a[i]; a[i] = a[j]; a[j] = temp;
        }
    }
    temp = a[i + 1]; a[i + 1] = a[high]; a[high] = temp;
    return i + 1;
}
int main() {
    int a[100], stack[200], n, i, top = -1, low, high, p;
    scanf("%d", &n);
    for (i = 0; i < n; i++) scanf("%d", &a[i]);
    if (n > 1) { stack[++top] = 0; stack[++top] = n - 1; }
    while (top >= 0) {
        high = stack[top--]; low = stack[top--];
        p = partition(a, low, high);
        if (p - 1 > low) { stack[++top] = low; stack[++top] = p - 1; }
        if (p + 1 < high) { stack[++top] = p + 1; stack[++top] = high; }
    }
    for (i = 0; i < n; i++) printf("%d ", a[i]);
    return 0;
}
