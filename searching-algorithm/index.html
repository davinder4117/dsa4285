1. Enter an integer array from the user, enter an integer value and perform linear search through iteration.

#include <stdio.h>

int main()
{
    int a[100], n, key, i, found = 0;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    if (n < 1 || n > 100)
    {
        printf("Invalid array size\n");
        return 1;
    }

    printf("Enter array elements:\n");
    for (i = 0; i < n; i++)
        scanf("%d", &a[i]);

    printf("Enter value to search: ");
    scanf("%d", &key);

    for (i = 0; i < n; i++)
    {
        if (a[i] == key)
        {
            printf("Element found at position %d\n", i + 1);
            found = 1;
            break;
        }
    }

    if (!found)
        printf("Element not found\n");

    return 0;
}

2. Enter a string from the user, enter a character and perform linear search through recursion.

#include <stdio.h>

int linearSearch(char str[], char key, int i)
{
    if (str[i] == '\0')
        return -1;

    if (str[i] == key)
        return i;

    return linearSearch(str, key, i + 1);
}

int main()
{
    char str[100], key;
    int pos;

    printf("Enter a string: ");
    scanf("%99s", str);

    printf("Enter character to search: ");
    scanf(" %c", &key);

    pos = linearSearch(str, key, 0);

    if (pos == -1)
        printf("Character not found\n");
    else
        printf("Character found at position %d\n", pos + 1);

    return 0;
}

5. Enter an integer array from the user, enter an integer value and perform binary search through iteration.

#include <stdio.h>

int main()
{
    int a[100], n, key, i;
    int low, high, mid, found = 0;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    if (n < 1 || n > 100)
    {
        printf("Invalid array size\n");
        return 1;
    }

    printf("Enter elements in sorted ascending order:\n");
    for (i = 0; i < n; i++)
        scanf("%d", &a[i]);

    printf("Enter value to search: ");
    scanf("%d", &key);

    low = 0;
    high = n - 1;

    while (low <= high)
    {
        mid = low + (high - low) / 2;

        if (a[mid] == key)
        {
            printf("Element found at position %d\n", mid + 1);
            found = 1;
            break;
        }
        else if (a[mid] < key)
            low = mid + 1;
        else
            high = mid - 1;
    }

    if (!found)
        printf("Element not found\n");

    return 0;
}

6. Enter a string from the user, enter a character and perform binary search through recursion.

#include <stdio.h>

int binarySearch(char str[], char key, int low, int high)
{
    int mid;

    if (low > high)
        return -1;

    mid = low + (high - low) / 2;

    if (str[mid] == key)
        return mid;
    else if (str[mid] < key)
        return binarySearch(str, key, mid + 1, high);
    else
        return binarySearch(str, key, low, mid - 1);
}

int main()
{
    char str[100], key;
    int pos, n = 0;

    printf("Enter a string in sorted character order: ");
    scanf("%99s", str);

    printf("Enter character to search: ");
    scanf(" %c", &key);

    while (str[n] != '\0')
        n++;

    pos = binarySearch(str, key, 0, n - 1);

    if (pos == -1)
        printf("Character not found\n");
    else
        printf("Character found at position %d\n", pos + 1);

    return 0;
}

9. Calculate the best and worst complexity of the linear searching algorithm on an integer array.

#include <stdio.h>

int main()
{
    int a[100], n, key, i;
    int comparisons = 0, pos = -1;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    if (n < 1 || n > 100)
    {
        printf("Invalid array size\n");
        return 1;
    }

    printf("Enter array elements:\n");
    for (i = 0; i < n; i++)
        scanf("%d", &a[i]);

    printf("Enter value to search: ");
    scanf("%d", &key);

    for (i = 0; i < n; i++)
    {
        comparisons++;

        if (a[i] == key)
        {
            pos = i;
            break;
        }
    }

    if (pos == -1)
        printf("Element not found\n");
    else
        printf("Element found at position %d\n", pos + 1);

    printf("Number of comparisons: %d\n", comparisons);
    printf("Best-case time complexity: O(1)\n");
    printf("Worst-case time complexity: O(n)\n");

    return 0;
}
