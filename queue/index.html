Question 1

Implement the operations of simple queue in an integer array (Insert, Delete, Peek, IsEmpty, IsFull).

Answer 1

#include <stdio.h>
#define SIZE 5

int queue[SIZE];
int front = -1, rear = -1;

int isEmpty()
{
    return front == -1;
}

int isFull()
{
    return rear == SIZE - 1;
}

void insert(int value)
{
    if (isFull())
        printf("Queue is full\n");
    else
    {
        if (front == -1)
            front = 0;
        queue[++rear] = value;
    }
}

void delete()
{
    if (isEmpty())
        printf("Queue is empty\n");
    else
    {
        printf("Deleted: %d\n", queue[front]);
        if (front == rear)
            front = rear = -1;
        else
            front++;
    }
}

void peek()
{
    if (isEmpty())
        printf("Queue is empty\n");
    else
        printf("Front: %d\n", queue[front]);
}

int main()
{
    insert(10);
    insert(20);
    insert(30);
    peek();
    delete();
    peek();

    printf("Empty: %d\n", isEmpty());
    printf("Full: %d\n", isFull());

    return 0;
}

Question 3

Implement the operations of a dynamic integer queue (Insert, Delete, Peek, IsEmpty, IsFull).

Answer 3

#include <stdio.h>
#include <stdlib.h>

struct Node
{
    int data;
    struct Node *next;
};

struct Node *front = NULL, *rear = NULL;

int isEmpty()
{
    return front == NULL;
}

int isFull()
{
    struct Node *temp = malloc(sizeof(struct Node));
    if (temp == NULL)
        return 1;
    free(temp);
    return 0;
}

void insert(int value)
{
    struct Node *temp = malloc(sizeof(struct Node));

    if (temp == NULL)
    {
        printf("Queue is full\n");
        return;
    }

    temp->data = value;
    temp->next = NULL;

    if (rear == NULL)
        front = rear = temp;
    else
    {
        rear->next = temp;
        rear = temp;
    }
}

void delete()
{
    if (isEmpty())
        printf("Queue is empty\n");
    else
    {
        struct Node *temp = front;
        printf("Deleted: %d\n", temp->data);
        front = front->next;

        if (front == NULL)
            rear = NULL;

        free(temp);
    }
}

void peek()
{
    if (isEmpty())
        printf("Queue is empty\n");
    else
        printf("Front: %d\n", front->data);
}

int main()
{
    insert(10);
    insert(20);
    insert(30);
    peek();
    delete();
    peek();

    printf("Empty: %d\n", isEmpty());
    printf("Full: %d\n", isFull());

    while (!isEmpty())
        delete();

    return 0;
}

Question 7

Implement the operations of a priority queue in an integer array (Insert, Delete, Peek, IsEmpty, IsFull).

Answer 7

#include <stdio.h>
#define SIZE 5

int queue[SIZE];
int count = 0;

int isEmpty()
{
    return count == 0;
}

int isFull()
{
    return count == SIZE;
}

void insert(int value)
{
    if (isFull())
    {
        printf("Queue is full\n");
        return;
    }

    int i = count - 1;

    while (i >= 0 && queue[i] > value)
    {
        queue[i + 1] = queue[i];
        i--;
    }

    queue[i + 1] = value;
    count++;
}

void delete()
{
    if (isEmpty())
        printf("Queue is empty\n");
    else
    {
        printf("Deleted: %d\n", queue[0]);

        for (int i = 0; i < count - 1; i++)
            queue[i] = queue[i + 1];

        count--;
    }
}

void peek()
{
    if (isEmpty())
        printf("Queue is empty\n");
    else
        printf("Highest priority: %d\n", queue[0]);
}

int main()
{
    insert(30);
    insert(10);
    insert(20);
    peek();
    delete();
    peek();

    printf("Empty: %d\n", isEmpty());
    printf("Full: %d\n", isFull());

    return 0;
}
